Ingresos totales = SUM(IngresosDiarios[Ingresos])
Número de pagos = SUM(IngresosDiarios[CantidadPagos])
Ticket promedio = DIVIDE([Ingresos totales], [Número de pagos], 0)
Reservas activas = SUM(RankingClases[ReservasActivas])
Sesiones programadas = COUNTROWS(OcupacionSesiones)
Cupos disponibles = SUM(OcupacionSesiones[CupoMaximo]) - SUM(OcupacionSesiones[ReservasActivas])
Ocupación ponderada % = DIVIDE(SUM(OcupacionSesiones[ReservasActivas]), SUM(OcupacionSesiones[CupoMaximo]), 0)
Variación ocupación p.p. = AVERAGE(OcupacionSesiones[VariacionPuntos])
Socios activos (total) = MAX(Indicadores[SociosActivos])
